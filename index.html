<script>

  // Code Generated by Sidekick is for learning and experimentation purposes only.


  let lists = [];

  let currentListId = null;


  // ── Load from localStorage ──

  function loadData() {

    const saved = localStorage.getItem('my_checklists');

    lists = saved ? JSON.parse(saved) : [];

  }


  function saveData() {

    localStorage.setItem('my_checklists', JSON.stringify(lists));

  }


  // ── Emoji picker based on name keywords ──

  function pickEmoji(name) {

    const n = name.toLowerCase();

    if (n.match(/trip|travel|pack|vac|flight|hotel/)) return '✈️';

    if (n.match(/recipe|cook|food|meal|dinner|lunch|breakfast/)) return '🍳';

    if (n.match(/shop|grocery|store|buy|market/)) return '🛒';

    if (n.match(/work|meeting|office|task|project/)) return '💼';

    if (n.match(/gym|workout|fitness|exercise/)) return '💪';

    if (n.match(/book|read|study|learn/)) return '📚';

    if (n.match(/movie|show|watch/)) return '🎬';

    if (n.match(/party|event|birthday|wedding/)) return '🎉';

    return '📋';

  }


  // ── Home Screen ──

  function renderHome() {

    const container = document.getElementById('lists-container');

    if (lists.length === 0) {

      container.innerHTML = `

        <div class="empty-state">

          <div class="icon">📭</div>

          <div>No checklists yet. Create one above!</div>

        </div>`;

      return;

    }


    container.innerHTML = lists.map(list => {

      const total = list.items.length;

      const done  = list.items.filter(i => i.checked).length;

      return `

        <div class="list-card" onclick="openList('${list.id}')">

          <span class="list-card-emoji">${list.emoji}</span>

          <div class="list-card-info">

            <div class="list-card-name">${list.name}</div>

            <div class="list-card-meta">${done} / ${total} items done</div>

          </div>

          <button class="btn-icon" onclick="deleteList(event,'${list.id}')">🗑</button>

        </div>`;

    }).join('');

  }


  function createList() {

    const input = document.getElementById('new-list-input');

    const name  = input.value.trim();

    if (!name) return;


    const list = {

      id:    Date.now().toString(),

      name,

      emoji: pickEmoji(name),

      items: []

    };

    lists.unshift(list);

    saveData();

    renderHome();

    input.value = '';

    input.focus();

  }


  function deleteList(e, id) {

    e.stopPropagation();

    if (!confirm('Delete this checklist?')) return;

    lists = lists.filter(l => l.id !== id);

    saveData();

    renderHome();

  }


  // ── List Screen ──

  function openList(id) {

    currentListId = id;

    document.getElementById('home-screen').classList.remove('active');

    document.getElementById('list-screen').classList.add('active');

    renderList();

    setTimeout(() => document.getElementById('new-item-input').focus(), 300);

  }


  function goHome() {

    currentListId = null;

    document.getElementById('list-screen').classList.remove('active');

    document.getElementById('home-screen').classList.add('active');

    renderHome();

  }


  function renderList() {

    const list = lists.find(l => l.id === currentListId);

    if (!list) return;


    document.getElementById('list-screen-title').textContent = `${list.emoji} ${list.name}`;


    const total   = list.items.length;

    const done    = list.items.filter(i => i.checked).length;

    const percent = total === 0 ? 0 : Math.round((done / total) * 100);


    document.getElementById('progress-fill').style.width  = percent + '%';

    document.getElementById('progress-label').textContent = `${done} of ${total} done`;


    const ul = document.getElementById('checklist-items');

    if (list.items.length === 0) {

      ul.innerHTML = `

        <div class="empty-state">

          <div class="icon">🗒️</div>

          <div>No items yet. Add something above!</div>

        </div>`;

      return;

    }


    ul.innerHTML = list.items.map((item, idx) => `

      <li class="checklist-item ${item.checked ? 'done' : ''}">

        <input type="checkbox" ${item.checked ? 'checked' : ''}

          onchange="toggleItem(${idx})"/>

        <span class="item-label">${item.text}</span>

        <button class="btn-icon" onclick="deleteItem(${idx})">✕</button>

      </li>`).join('');

  }


  function addItem() {

    const input = document.getElementById('new-item-input');

    const text  = input.value.trim();

    if (!text) return;


    const list = lists.find(l => l.id === currentListId);

    list.items.push({ text, checked: false });

    saveData();

    renderList();

    input.value = '';

    input.focus();

  }


  function toggleItem(idx) {

    const list = lists.find(l => l.id === currentListId);

    list.items[idx].checked = !list.items[idx].checked;

    saveData();

    renderList();

  }


  function deleteItem(idx) {

    const list = lists.find(l => l.id === currentListId);

    list.items.splice(idx, 1);

    saveData();

    renderList();

  }


  function clearChecked() {

    const list = lists.find(l => l.id === currentListId);

    if (!list.items.some(i => i.checked)) return;

    if (!confirm('Remove all checked items?')) return;

    list.items = list.items.filter(i => !i.checked);

    saveData();

    renderList();

  }


  // ── Enter key support ──

  document.getElementById('new-list-input').addEventListener('keydown', e => {

    if (e.key === 'Enter') createList();

  });

  document.getElementById('new-item-input').addEventListener('keydown', e => {

    if (e.key === 'Enter') addItem();

  });


  // ── Init ──

  loadData();

  renderHome();

</script>
